\newcommand{\checkfontsize}[2]{%
  \ifdim\f@size pt=#2\relax\else
    \PackageError{table-font}{#1: unexpected font size \f@size}{}%
  \fi
}

% Trial passes must not create PDF structures that survive counter rollback.
\AddToHook{begindocument/end}[table-font/trial-check]{%
  \let\test@TXtrial\TX@trial
  \def\TX@trial#1{%
    \test@TXtrial{#1%
      \csname tag_if_active:T\endcsname{%
        \PackageError{table-font}{Tagging active during tabularx trial}{}%
      }%
    }%
  }%
}
\DeclareHookRule{begindocument/end}{table-font/trial-check}{after}{neudissertation/tabularx-tagging}

% Legacy captions need the structure belonging to this [H] float, not an
% empty value or the parent left over from an earlier float.
\newcommand{\checklegacyfloatparent}{%
  \IfPackageAtLeastTF{latex-lab-testphase-float}{2024/01/01}{}{%
    \ifx\@current@float@struct\@empty
      \PackageError{table-font}{Missing legacy H float parent}{}%
    \else
      \ifnum\@current@float@struct>0\relax\else
        \PackageError{table-font}{Invalid legacy H float parent}{}%
      \fi
      \ifcsname test@previousfloatparent\endcsname
        \ifx\@current@float@struct\test@previousfloatparent
          \PackageError{table-font}{Reused legacy H float parent}{}%
        \fi
      \fi
      \global\let\test@previousfloatparent\@current@float@struct
    \fi
  }%
}
