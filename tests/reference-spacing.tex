\newsavebox{\actualref}
\newsavebox{\expectedref}
\newcommand{\checkrefspacing}[3]{%
  \sbox{\actualref}{#2}%
  \sbox{\expectedref}{#3}%
  \ifdim\dimexpr\wd\actualref-\wd\expectedref\relax>0.01pt
    \PackageError{reference-spacing}{#1 too wide}{}%
  \fi
  \ifdim\dimexpr\wd\expectedref-\wd\actualref\relax>0.01pt
    \PackageError{reference-spacing}{#1 too narrow}{}%
  \fi
}
